25. CONVERSION FUNCTION

function convertPrice(fcfaAmount, targetCurrency) {
  const rate = rates[targetCurrency] || 1;
  const converted = fcfaAmount * rate;
  if (targetCurrency === 'FCFA') {
    return new Intl.NumberFormat('fr-FR').format(fcfaAmount) + ' FCFA';
  }
  return new Intl.NumberFormat('fr-FR', {
    minimumFractionDigits: 2, maximumFractionDigits: 2
  }).format(converted) + ' ' + targetCurrency;
}
